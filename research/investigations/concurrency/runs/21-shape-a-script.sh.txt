#!/bin/sh
# One-shot: shape A (proto_style A 2, which calls selector_matches for every rule and element) on ecma262,
# sequential drivers built from ec21b7c (before sibling positions) and 95b9073 (positions passed as a slice), three runs each.
set -eu
d=build/research/concurrency; ua=research/investigations/concurrency/ua.css; S=$1
for i in 1 2 3; do
  for pair in "ec21b7c:$S/proto_style_idx_seq" "95b9073:build/proto_style_seq"; do
    name=${pair%%:*}; bin=${pair#*:}
    t=$(command time -p sh -c 'exec "$@" >/dev/null 2>&1' sh "$bin" A 2 $d/ecma262.html $ua $d/ecma262-ecmarkup.css $d/ecma262-print.css 2>&1 | awk '$1=="real"{print $2}')
    echo "run $i $name $t"
  done
done

#!/bin/sh
# One-shot: shape C's style stage with the rule index alone (ec21b7c) and with sibling positions too (the commit after aacd878), same run.
set -eu
d=build/research/concurrency; ua=research/investigations/concurrency/ua.css; S=$1
best() { b=$1; w=$2; r=$3; shift 3; least=; for i in 1 2 3; do t=$(WF_WORKERS=$w command time -p sh -c 'exec "$@" >/dev/null 2>&1' sh "$b" C "$r" "$@" 2>&1 | awk '$1=="real"{print $2}'); if [ -z "$least" ] || [ "$(echo "$t < $least" | bc)" = 1 ]; then least=$t; fi; done; echo "$least"; }
echo "page driver workers reps T(0) T(REPS) stage"
for p in ecma262 html5 apollo11; do
  case $p in ecma262) sh="$d/ecma262-ecmarkup.css $d/ecma262-print.css";; apollo11) sh="$d/apollo11-modules.css $d/apollo11-site.css";; *) sh="";; esac
  for pair in "index-par:$S/proto_style_idx_par:4" "positions-par:build/proto_style:4" "index-seq:$S/proto_style_idx_seq:1" "positions-seq:build/proto_style_seq:1"; do
    name=${pair%%:*}; rest=${pair#*:}; bin=${rest%%:*}; w=${rest##*:}
    z=$(best "$bin" "$w" 0 $d/$p.html $ua $sh); f=$(best "$bin" "$w" 5 $d/$p.html $ua $sh)
    echo "$p $name $w 5 $z $f" | awk '{ printf "%s %s %s %s %s %s %.4f\n", $1, $2, $3, $4, $5, $6, ($6 - $5) / $4 }'
  done
done
